from fastapi import FastAPI
from LizCore import ModelInferenceEngine

app = FastAPI(title="Liz Inspired API")
engine = ModelInferenceEngine(weights="prod_v2.pt")

@app.post("/v1/predict")
async def run_inference(payload: dict):
    prediction = await engine.process(payload)
    return {"status": 200, "result": prediction}
